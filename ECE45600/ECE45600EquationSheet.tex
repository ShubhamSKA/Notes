\section*{Inverter Chain}
\pagestyle{empty}
\begin{equation*}
    F = \frac{C_L}{C_{g_1}}:~~ C_L=\text{Load cap, } C_{g_1}=\text{Internal cap}
\end{equation*}
\begin{equation*}
    f = \sqrt[N]{F}:~~ f=\text{scaling factor, } N=\text{\# inverters}
\end{equation*}
\begin{equation*}
    t_p = Nt_{p_0}\left(1+\frac{f}{\gamma}\right):~~t_p=\text{prop delay}, t_{p_0} = \text{single  prop delay}
\end{equation*}
\begin{equation*}
    \gamma = \frac{\text{intrinsic capacitance at output}}{\text{input gate capacitance}}
\end{equation*}
\begin{equation*}
    \sum C_L = \sum_{k=0}^{N} (f^N+\gamma f^N)C_{g_1}+C_L = C_{g_1}((1+\gamma)(1+f)+F)
\end{equation*}
Note: the formula above includes the input capcacitance.
\begin{equation*}
    P_{diss} = (\textbf{freq})\alpha_{0\rightarrow 1}V_{DD}^2\sum C_L:~~ \alpha_{0\rightarrow 1}=\text{activity factor}
\end{equation*}
\section*{Formula to Circuit}
\begin{enumerate}
    \item Get formula to the form $\bar{X}$
    \item The pull-up of the circuit will be composed entirely of PMOS:
          \begin{itemize}
              \item AND: parallel
              \item OR: series
          \end{itemize}
          The pull-down will be the inverse, that is NMOS:
          \begin{itemize}
              \item AND: series
              \item OR: parallel
          \end{itemize}
    \item To scale the transistors, the initial scale will be that of the minimized
          inverter. The multiplication factor of the transistors will be such that the resistance of each path in each netweork is the same as that of the corresponding path in an inverter.\\
          $C_{unit}$ is sum of the scaling of the PMOS and the NMOS for the given input.
\end{enumerate}
