\section*{SinCos, etc.}
\begin{align*}
    \cos(x)=\frac{e^{jx}+e^{-jx}}{2} \\
    \sin(x)=\frac{e^{jx}-e^{-jx}}{2j}
\end{align*}
\begin{equation*}
    \omega = \frac{2\pi}{T}= \frac{2\pi}{N}
\end{equation*}
\section*{LTI}
Linear: $\alpha_1 x_1(t)+\alpha_2 x_2(t) = \alpha_1 y_1(t)+\alpha_2 y_2(t) ~\forall~ \alpha_1,\alpha_2$\\
Time invariant: $x(t) \rightarrow y(t) \implies x(t-t_0) \rightarrow y(t-t_0)$
\section*{Convolution}
Convolution:
\begin{equation*}
    y=\sum_{k=-\infty}^{\infty}x[k]h[n-k]=\int_{-\infty}^{\infty}x(\tau)h(t-\tau)d\tau
\end{equation*}
Impulse Response: $h=S[\delta[n]]=S(\delta(t)) = \frac{d}{dt}S(u(t))$
\subsection*{Properties}
$x*h=h*x$\\
$x*(h_1+h_2)=x*h_1+x*h_2$\\
$x*(h_1*h_2)=(x*h_1)*h_2$
\subsection*{Classifications}
Memoryless: $h[k]=0=h(t)~\forall~k,t\neq 0$\\ Invertible: $\exists h_I |
    h*h_I=\delta$\\ Causal: $h[k]=h(t)=0 ~\forall~ k,t<0$\\ Stable:
$\sum_{-\infty}^{\infty}|h[k]|<\infty$,$\int_{-\infty}^{\infty}h(\tau)d\tau<\infty$
\section*{Fourier}
\begin{align*}
    y(t)=e^{j\omega t}*h(t)=e^{j\omega t}\int_{-\infty}^{\infty}h(\tau)e^{-j\omega \tau}d\tau \\
    =e^{j\omega t}|H(j\omega)|e^{j\phase{H}(j\omega)}
\end{align*}
\begin{equation*}
    a_k = \frac{1}{T} \int_{<T>} x(t) e^{-jk\omega_0 t} dt = \frac{1}{N} \sum_{n=<N>}x[n]e^{-jk\frac{2\pi}{N} n}
\end{equation*}
\begin{equation*}
    x=\sum_{-\infty}^{\infty}a_ke^{jk\omega_0t} = \sum_{k=<N>}a_k x_k[n]
\end{equation*}
Real $x(t)$: $a_k=a^*_{-k}$\\
Even $x(t)$: $a_k=a_{-k}$
