\section*{Classifications}
\pagestyle{empty}
\textbf{Linear}: $\alpha_1 x_1(t)+\alpha_2 x_2(t) = \alpha_1 y_1(t)+\alpha_2 y_2(t) ~\forall~ \alpha_1,\alpha_2$\\
To prove, sum first, then transform, and prove equal to, transform first then sum.\\
\textbf{Energy}:
\begin{equation*}
    E = \int_{t_1}^{t_2} |x(t)|^2 dt = \int_{t_1}^{t_2}(x^2_{Re}(t)+x^2_{Im}(t))dt
\end{equation*}
for total energy, $t_1=-\infty, t_2 = \infty$.\\
\textbf{Power}:
\begin{equation*}
    P_{avg} = \frac{1}{t_2-t_1}E = \frac{1}{t_2-t_1}\int_{t_1}^{t_2}(x^2_{Re}(t)+x^2_{Im}(t))dt
\end{equation*}
(In discrete time, the denominator is $n_2-n_1+1$). For total, both limits tend to $\infty$.\\
\textbf{Period}: $\exists T \neq 0 | x(t) = x(t+T) ~\forall t \implies \text{periodic}$.\\
Minimum T that fulfills conditions is $T_0$, fundamental period.
For two added signals, $T = LCM(T_1,T_2)$.\\
$T(\sin(\omega t)) = \frac{2\pi}{\omega}$\\
\textbf{Parity}:\\
\underline{Even (+)}: $x(t)=x(-t)$\\
\underline{Odd (-)}: $x(t) = -x(-t)$\\
\begin{equation*}
    x = even+odd = \frac{x(t)+x(-t)}{2}+\frac{x(t)-x(-t)}{2}
\end{equation*}
\textbf{Complex Exponential}:
\begin{align*}
    x(t) = Ce^{\alpha t} & = |C|\cdot e^{\sigma t}\cdot e^{j(\omega t +\phi)}                       \\
                         & = |C|\cdot e^{\sigma t}\cdot (\cos(\omega t +\phi)+\sin(\omega t +\phi))
\end{align*}
$\omega = 2\pi/T$ is the fundamental frequency.
\begin{align*}
    P(t) = |C|^2w^{2\sigma t} \\
    E_\infty = \infty , P_\infty = C^2
\end{align*}
\underline{HRCEs}: $x_k(t) = e^{jk\omega_0 t}$
\section*{Transformations}
\textbf{Time Shift}:$x(t)\rightarrow x(t-t_0)$.Moves signal right by $t_0$.\\
\textbf{Time Reversal}:$x(t)\rightarrow x(-t)$.Reflects signal about y-axis.\\
\textbf{Time Scale}:$x(t)\rightarrow x(\alpha t)$.Widens signal by $\frac{1}{\alpha}$.
\section*{Complex Exponential}
\textbf{DTCE}:
$x[n] = |C|\cdot e^{\sigma n} \cdot e^{j\omega n +\phi}$\\
Has exponential envelope $|C|e^{\sigma t}$.\\
\begin{equation*}
    N_0 = LCM\left(T = \frac{2\pi}{\omega}, 1\right)
\end{equation*}
For sum, $N = LCM(N_1,N_2)$.\\
If no $\pi$ in $\omega$, or $\omega$ is irrational, the signal is not periodic.
$0\leq \omega < 2\pi$.
\\Speed up for $\omega\approx\pm\pi$, slow at borders, that is for $k=\lfloor\frac{N}{2}\rfloor  or \lceil\frac{N}{2}\rceil$.\\
\underline{DTHRCEs}: $x_k[n] = e^{jk(\frac{2\pi}{N})n}$\\
Only N distinct members in the HRCE: $k = 0 \ldots N-1$.
\section*{Special Signals}
\textbf{DT}:\\
$\delta[n] = 0 ~\forall~ n \neq 0,\text{ else } 1$.\\
$u[n] =\sum_{k=0}^{\infty}\delta[n-k] = 1 ~\forall~ n\geq 0$\\
$x[n] = \sum_{k=-\infty}^{\infty}x[k]\delta[n-k]$\\
\textbf{CT}:\\
$\int_{-\infty}^{\infty}\delta(t) = 1$

\section*{Properties}
\textbf{Memory}: if $y(t) = f(x(T)) | T\neq t_0$, the system has memory.\\
\textbf{Invertible}: $x\rightarrow [S]\rightarrow [S^{-1}]\rightarrow x$
\begin{equation*}
    \forall x_1, x_2 | x_1\neq x_2 \implies y_1\neq y_2
\end{equation*}
Can be proven if $x$ as a function of $y$ is defined across all $t$.\\
\textbf{Causal}: $y(t) = \sum_{T=-\infty}^{t} f(x(T))$. No dependence on future values of t.\\
\textbf{Stable}: $|x(t)|<B \implies |y(t)|<M$ for finite B, M.\\
\textbf{Time Invariant}: $x(t) \rightarrow y(t) \implies x(t-t_0) \rightarrow y(t-t_0)$.
Scaling, or having a function of $t$ outside, like ($tx(t)$), will \textbf{NOT} be time invariant.

\end{align*}

\section*{Fourier}
\begin{align*}
    y(t)=e^{j\omega t}*h(t)=e^{j\omega t}\int_{-\infty}^{\infty}h(\tau)e^{-j\omega \tau}d\tau
    =e^{j\omega t}|H(j\omega)|e^{j\phase{H}(j\omega)} =\sum_\omega \mathbf{x(e^{j\omega t})H(j\omega)}
\end{align*}
\begin{equation*}
    a_k = \frac{1}{T} \int_{<T>} x(t) e^{-jk\omega_0 t} dt =\frac{X(j\omega)}{T}= \frac{1}{N} \sum_{n=<N>}x[n]e^{-jk\frac{2\pi}{N} n} = \frac{X(e^{jk\omega_0})}{N}
\end{equation*}
\begin{equation*}
    x(t)=\sum_{-\infty}^{\infty}a_ke^{jk\omega_0t}=\frac{1}{2\pi}\int_{-\infty}^{\infty}X(j\omega)e^{j\omega t}d\omega ~,~
    x[n]=\sum_{k=<N>}a_k e^{jk\omega_0 n} =  \frac{1}{2\pi}\int_{2\pi} X(e^{j\omega})e^{j\omega n}d\omega
\end{equation*}
\begin{eqnarray*}
    X(j\omega) = 2\pi \sum_{k=-\infty}^{\infty}a_k\delta (\omega - \frac{2\pi K}{T}) = \int_{-\infty}^{\infty} x(t)e^{-j\omega t}dt ~,~ X(e^{j\omega}) = 2\pi \sum_{k=-\infty}^{\infty}a_k \delta(\omega - \frac{2\pi K}{N}) = \sum_{n=-\infty}^{\infty}x[n]e^{-j\omega n}
\end{eqnarray*}
\begin{equation*}
    \sum_{m=-\infty}^{n}x[m]\rightarrow \frac{1}{1-e^{-j\omega}}X(e^{j\omega})+\pi X(e^{j0})\sum_{k=-\infty}^{\infty}\delta(\omega-2\pi k)
\end{equation*}
Real $x(t)$: $a_k=a^*_{-k}$\\
Even $x(t)$: $a_k=a_{-k}$\\
Duality: $\mathbf{F}\{X(t)\} = 2\pi\cdot x(-\omega)$ \\
Convergence: $E_\infty < \infty$

$x(t)\rightarrow [H_1]\rightarrow[H_2]\rightarrow y(t) = x(t)\rightarrow[H_1\cdot H_2]\rightarrow y(t)$\\

$H(j\omega) = \frac{Y(j\omega)}{X(j\omega)}$
\section*{Diff}
Nth order linear constant diffEq:
\begin{equation*}
    \sum_{k=0}^{N}a_ky[n-k] = \sum_{k=0}^{M}b_kx[n-k] \implies H(e^{j\omega}) = \frac{\sum_{k=0}^{M} b_ke^{-jk\omega}}{\sum_{k=0}^{N} a_ke^{-jk\omega}}
\end{equation*}
\begin{equation*}
    \sum_{k=0}^{N}a_k\frac{d^ky(t)}{dt^k} = \sum_{k=0}^{M}b_k\frac{d^kx(t)}{dt^k} \implies H(j\omega) = \frac{\sum_{k=0}^{M}b_k(j\omega)^k}{\sum_{k=0}^{N}a_k(j\omega)^k}
\end{equation*}

\section*{Sampling}
Sampling is done using an impulse train, that is:
\begin{align*}
    p(t) = \sum_{n=-\infty}^{\infty}\delta(t-nT)                                   \\
    P(j\omega) = \frac{2\pi}{T}\sum_{k=-\infty}^{\infty}\delta(\omega - k\omega_s) \\
    X_p(j\omega) = \frac{1}{T}\sum_{k=-\infty}^{\infty} X(j\omega-jk\omega_s)
\end{align*}
For no overlap: $\omega_s = \frac{2\pi}{T} > 2\omega_M$
